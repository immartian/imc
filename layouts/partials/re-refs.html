{{- /* Turns {ch4}, {M}, {P} ... in already-rendered HTML into links to the RE thesis. */ -}} {{- $s := . -}} {{- $d := hugo.Data.continuum -}} {{- range $key, $c := $d.chapters -}} {{- $link := printf `%s` $d.re_base $c.file $c.title $c.label -}} {{- $s = replace $s (printf "{%s}" $key) $link -}} {{- end -}} {{- $s | safeHTML -}}
